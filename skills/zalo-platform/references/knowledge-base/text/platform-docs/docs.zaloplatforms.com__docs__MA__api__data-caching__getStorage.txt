Sample Code​
import { getStorage } from "zmp-sdk/apis";
getStorage({
keys: ["key1", "key2", "key3"],
success: (data) => {
// xử lý khi gọi api thành công
const { key1, key2, key3 } = data;
},
fail: (error) => {
// xử lý khi gọi api thất bại
console.log(error);
},
});

import { getStorage } from "zmp-sdk/apis";
const getData = async () => {

keys: ["key1", "key2", "key3"],
});
} catch (error) {
// xử lý khi gọi api thất bại
console.log(error);
}
};
